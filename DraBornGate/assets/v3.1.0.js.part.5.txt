  for (const dkdButton of document.querySelectorAll('[data-dkd-v31-more]')) {
    dkdButton.addEventListener('click', () => {
      const dkdKey = dkdButton.dataset.dkdV31More;
      dkdV31State.queueLimits[dkdKey] = (dkdV31State.queueLimits[dkdKey] || 3) + 5;
      dkdV31RenderSimple();
      document.querySelector(`[data-dkd-v31-category="${dkdKey}"]`)?.setAttribute('open', '');
    });
  }
  for (const dkdButton of document.querySelectorAll('[data-dkd-v31-pass]')) {
    dkdButton.addEventListener('click', () => {
      const dkdPass = dkdV31State.queue.find((dkdRow) => dkdRow.pass_id === dkdButton.dataset.dkdV31Pass);
      if (dkdPass) dkdV31OpenPassModal(dkdPass, dkdPass.approval_code);
    });
  }
  document.querySelector('#dkd-v31-retry-queue')?.addEventListener('click', dkdV31RefreshQueue);
}

function dkdV31EnsureModal() {
  let dkdModal = document.querySelector('#dkd-v31-global-modal');
  if (!dkdModal) {
    dkdModal = document.createElement('div');
    dkdModal.id = 'dkd-v31-global-modal';
    dkdModal.hidden = true;
    document.body.appendChild(dkdModal);
  }
  return dkdModal;
}

function dkdV31CloseModal() {
  const dkdModal = dkdV31EnsureModal();
  dkdModal.hidden = true;
  dkdModal.innerHTML = '';
  document.body.classList.remove('dkd-v31-modal-open');
  dkdV31GlobalModalMode = '';
}
