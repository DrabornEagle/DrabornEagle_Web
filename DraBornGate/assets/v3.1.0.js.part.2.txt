  if (Number.isNaN(dkdDate.getTime())) return String(dkdValue);
  return dkdDate.toLocaleString('tr-TR', { dateStyle: 'short', timeStyle: 'short' });
}

function dkdV31FormatMoney(dkdValue) {
  return new Intl.NumberFormat('tr-TR', { style: 'currency', currency: 'TRY', maximumFractionDigits: 2 }).format(Number(dkdValue || 0));
}

function dkdV31StatusLabel(dkdStatus) {
  return ({ arrived: 'Kapıda', waiting: 'Yaklaşıyor', approved: 'Onay Bekliyor', completed: 'Tamamlandı' })[dkdStatus] || String(dkdStatus || 'Aktif');
}

  return `<div class="dkd-v31-detail ${dkdWide ? 'wide' : ''}"><span>${dkdV31Icon(dkdIcon)}</span><div><small>${dkdV31.escape(dkdLabel)}</small><strong>${dkdV31.escape(dkdValue || '—')}</strong></div></div>`;
}
