function dkdV31PatchBottomCourierLabel() {
  const dkdNavigationCandidates = [...document.querySelectorAll('nav,[class*="bottom"],[class*="tab-bar"],[class*="navigation"]')];
  for (const dkdNavigation of dkdNavigationCandidates) {
    const dkdText = dkdV31.normalize(dkdNavigation.textContent);
    if (!dkdText.includes('yeni') || !dkdText.includes('gecislerim') || !dkdText.includes('hareket')) continue;

      .find((dkdElement) => dkdV31.normalize(dkdElement.textContent) === 'dbg');
