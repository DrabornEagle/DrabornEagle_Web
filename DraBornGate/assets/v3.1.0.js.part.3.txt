function dkdV31QueueCard(dkdPass) {
  return `<article class="dkd-v31-queue-card">
    ${dkdV31PassDetailsHtml(dkdPass, true)}
    <button type="button" class="dkd-v31-card-open" data-dkd-v31-pass="${dkdV31.escape(dkdPass.pass_id)}">${dkdV31Icon('search')}<span>Tüm Bilgileri Aç</span></button>
  </article>`;
}

function dkdV31QueueCategory(dkdKey, dkdTitle, dkdRows, dkdOpen = false) {
  const dkdLimit = dkdV31State.queueLimits[dkdKey] || 3;
  const dkdVisible = dkdRows.slice(0, dkdLimit);
  const dkdRemaining = Math.max(0, dkdRows.length - dkdVisible.length);
  return `<details class="dkd-v31-queue-category" data-dkd-v31-category="${dkdKey}" ${dkdOpen ? 'open' : ''}>
    <summary><div><strong>${dkdV31.escape(dkdTitle)}</strong><small>${dkdRows.length} kurye</small></div>${dkdV31Icon('chevron')}</summary>
    <div class="dkd-v31-category-body">
      ${dkdVisible.length ? dkdVisible.map(dkdV31QueueCard).join('') : '<div class="dkd-v31-empty-mini">Bu kategoride aktif kurye yok.</div>'}
      ${dkdRemaining ? `<button type="button" class="dkd-v31-more" data-dkd-v31-more="${dkdKey}">${dkdV31Icon('plus')}<span>5 Tane Daha Göster</span><b>${dkdRemaining} kayıt kaldı</b></button>` : ''}
    </div>
  </details>`;
}
